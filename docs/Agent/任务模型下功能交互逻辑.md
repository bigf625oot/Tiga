class TaskStatus(str, Enum):
    PENDING = "pending"
    PLANNING = "planning"     # 规划中（高亮虚线框）
    RUNNING = "running"      # 执行中（呼吸灯+Loading）
    VERIFYING = "verifying"  # 校验中（黄色快闪）
    COMPLETED = "completed"  # 已完成（绿色打勾+震动动画）
    FAILED = "failed"        # 失败（红色撤回）

class TaskStep(BaseModel):
    id: str                  # 唯一标识，如 "task_1"
    title: str
    status: TaskStatus
    progress: int            # 0-100 细粒度进度
    log_stream: List[str]    # 该任务产生的实时日志堆栈
